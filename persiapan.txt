-- 1. USERS (Admin Auth)
CREATE TABLE users (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    avatar_url VARCHAR(255) NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL
);

-- 2. PROFILE / SETTINGS (Hero & Bio Data)
CREATE TABLE profile (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(255) NOT NULL,
    tagline VARCHAR(255) NOT NULL,
    bio TEXT NOT NULL,
    email VARCHAR(255) NOT NULL,
    github_url VARCHAR(255) NULL,
    gitlab_url VARCHAR(255) NULL,
    linkedin_url VARCHAR(255) NULL,
    resume_pdf_path VARCHAR(255) NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL
);

-- 3. CATEGORIES (Kategori Proyek & Blog)
CREATE TABLE categories (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    slug VARCHAR(255) NOT NULL UNIQUE,
    type ENUM('project', 'post') NOT NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL
);

-- 4. PROJECTS (Portofolio Karya)
CREATE TABLE projects (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    category_id BIGINT NULL,
    title VARCHAR(255) NOT NULL,
    slug VARCHAR(255) NOT NULL UNIQUE,
    thumbnail VARCHAR(255) NOT NULL,
    short_description TEXT NOT NULL,
    full_description LONGTEXT NOT NULL,
    demo_url VARCHAR(255) NULL,
    repo_url VARCHAR(255) NULL,
    featured BOOLEAN DEFAULT FALSE,
    status ENUM('draft', 'published') DEFAULT 'draft',
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,
    FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE SET NULL
);

-- 5. SKILLS (Tech Stack)
CREATE TABLE skills (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    category ENUM('backend', 'frontend', 'database', 'tools') NOT NULL,
    icon_svg_or_class VARCHAR(255) NULL,
    order_num INT DEFAULT 0,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL
);

-- 6. PROJECT_SKILLS (Pivot Table Many-to-Many)
CREATE TABLE project_skills (
    project_id BIGINT NOT NULL,
    skill_id BIGINT NOT NULL,
    PRIMARY KEY (project_id, skill_id),
    FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE,
    FOREIGN KEY (skill_id) REFERENCES skills(id) ON DELETE CASCADE
);

-- 7. EXPERIENCES (Kerja, Magang, Edukasi)
CREATE TABLE experiences (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    company_or_client VARCHAR(255) NOT NULL,
    role VARCHAR(255) NOT NULL,
    type ENUM('fulltime', 'freelance', 'internship', 'education') NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NULL, -- NULL kalau masih berjalan (Present)
    description TEXT NOT NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL
);

-- 8. CERTIFICATES (Sertifikat & Lisensi)
CREATE TABLE certificates (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    experience_id BIGINT NULL, -- Opsional: kalau sertifikat nempel ke riwayat magang/kerja
    title VARCHAR(255) NOT NULL,
    issuer VARCHAR(255) NOT NULL,
    credential_id VARCHAR(255) NULL,
    credential_url VARCHAR(255) NULL,
    file_path VARCHAR(255) NULL, -- Path file PDF/Gambar di storage
    issue_date DATE NOT NULL,
    expiration_date DATE NULL, -- NULL kalau seumur hidup
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,
    FOREIGN KEY (experience_id) REFERENCES experiences(id) ON DELETE SET NULL
);

-- 9. POSTS (Blog / Artikel Teknis)
CREATE TABLE posts (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    category_id BIGINT NULL,
    title VARCHAR(255) NOT NULL,
    slug VARCHAR(255) NOT NULL UNIQUE,
    thumbnail VARCHAR(255) NULL,
    content LONGTEXT NOT NULL,
    status ENUM('draft', 'published') DEFAULT 'draft',
    published_at DATETIME NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,
    FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE SET NULL
);

-- 10. MESSAGES (Inbox Form Kontak)
CREATE TABLE messages (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    sender_name VARCHAR(255) NOT NULL,
    sender_email VARCHAR(255) NOT NULL,
    subject VARCHAR(255) NULL,
    message TEXT NOT NULL,
    is_read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL
);

1. Role-Based Access Control (RBAC):

   - Public Write Access: Allowed ONLY for creating documents in `messages` collection (sandboxed, read-restricted).
2. Environment Variables & Secret Handling:
   - Public client SDK MUST ONLY use `APPWRITE_ENDPOINT` and `PROJECT_ID`.
   - Secret API Keys with elevated admin privileges MUST NEVER be exposed to the frontend build and must reside solely inside Appwrite Go Functions.
3. Form Defense & Rate Limiting:
   - Contact form handler (Go Function) must implement IP-based rate limiting (max 3 submissions per 10 minutes) and input sanitization (XSS prevention) before persisting data to `messages`.
